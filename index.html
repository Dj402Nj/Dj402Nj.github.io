function showReportInApp() { try { const transactions = getData('transactions'); const rents = getData('rents'); const faezeh = getData('faezeh'); const personal = getData('personal'); const loans = getData('loanItems'); const savings = getData('savings'); const tasks = getData('tasks'); const invest = getData('invest'); let txt = '═══════════════════════════════════════════════════\n'; txt += ' 📊 گزارش کامل مه‌دیار\n'; txt += '═══════════════════════════════════════════════════\n\n'; // ====== تراکنش‌ها ====== txt += '💰 تراکنش‌ها:\n'; txt += '─────────────────────────────────────────────────\n'; if (transactions.length === 0) { txt += 'هیچ تراکنشی ثبت نشده\n'; } else { transactions.forEach(t => { txt += `📅 ${t.persianDate || 'نامشخص'} | ${t.type === 'income' ? '➕ درآمد' : '➖ خرج'} | ${t.amount.toLocaleString()} ریال | ${t.description || ''}\n`; }); } txt += '\n'; // ====== اجاره ====== txt += '🏠 اجاره:\n'; txt += '─────────────────────────────────────────────────\n'; if (rents.length === 0) { txt += 'هیچ اجاره‌ای ثبت نشده\n'; } else { rents.forEach(r => { txt += `📅 ${r.month || 'نامشخص'} | ${r.amount.toLocaleString()} ریال | ${r.paid ? '✅ پرداخت شد' : '❌ پرداخت نشده'}\n`; }); } txt += '\n'; // ====== فائزه ====== txt += '👩 فائزه:\n'; txt += '─────────────────────────────────────────────────\n'; if (faezeh.length === 0) { txt += 'هیچ موردی ثبت نشده\n'; } else { faezeh.forEach(f => { txt += `📅 ${f.month || 'نامشخص'} | ${f.amount.toLocaleString()} ریال\n`; }); } txt += '\n'; // ====== شخصی ====== txt += '🧑 مخارج شخصی:\n'; txt += '─────────────────────────────────────────────────\n'; if (personal.length === 0) { txt += 'هیچ موردی ثبت نشده\n'; } else { personal.forEach(p => { const emoji = { 'خوراک': '🍔', 'پوشاک': '👕', 'تفریح': '🎮', 'بهداشت': '🧴', 'سایر': '📦' }; txt += `📅 ${p.month || 'نامشخص'} | ${p.amount.toLocaleString()} ریال | ${emoji[p.type] || '📦'} ${p.type || 'سایر'}\n`; }); } txt += '\n'; // ====== اقساط ====== txt += '💰 اقساط:\n'; txt += '─────────────────────────────────────────────────\n'; if (loans.length === 0) { txt += 'هیچ قسطی ثبت نشده\n'; } else { loans.forEach(l => { txt += `📅 ${l.month || 'نامشخص'} | ${l.name || ''} | ${l.amount.toLocaleString()} ریال | ${l.paid ? '✅ پرداخت شد' : '❌ پرداخت نشده'}\n`; }); } txt += '\n'; // ====== پس‌انداز ====== txt += '🏦 پس‌انداز:\n'; txt += '─────────────────────────────────────────────────\n'; if (savings.length === 0) { txt += 'هیچ پس‌اندازی ثبت نشده\n'; } else { savings.forEach(s => { txt += `📅 ${toPersianDate(s.date)} | +${s.amount.toLocaleString()} ریال\n`; }); } txt += '\n'; // ====== کارها ====== txt += '✅ کارها:\n'; txt += '─────────────────────────────────────────────────\n'; if (tasks.length === 0) { txt += 'هیچ کاری ثبت نشده\n'; } else { tasks.forEach(t => { txt += `📅 ${t.date || 'نامشخص'} | ${t.text || ''} | ${t.done ? '✅ انجام شده' : '❌ انجام نشده'}\n`; }); } txt += '\n'; // ====== سرمایه‌گذاری ====== txt += '📈 سرمایه‌گذاری:\n'; txt += '─────────────────────────────────────────────────\n'; if (invest.length === 0) { txt += 'هیچ سرمایه‌گذاری ثبت نشده\n'; } else { invest.forEach(i => { const emoji = i.category === 'طلا' ? '🥇' : i.category === 'نقره' ? '🥈' : '₿'; txt += `📅 ${i.persianDate || 'نامشخص'} | ${emoji} ${i.category || ''} | ${i.amount.toLocaleString()} ریال | ${i.description || ''}\n`; }); } txt += '\n'; // ====== خلاصه مالی ====== const totalIncome = transactions.filter(t => t.type === 'income').reduce((s, t) => s + t.amount, 0); const totalExpense = transactions.filter(t => t.type === 'expense').reduce((s, t) => s + t.amount, 0); const totalRentPaid = rents.filter(r => r.paid).reduce((s, r) => s + r.amount, 0); const totalFaezeh = faezeh.reduce((s, f) => s + f.amount, 0); const totalPersonal = personal.reduce((s, p) => s + p.amount, 0); const totalLoanPaid = loans.filter(l => l.paid).reduce((s, l) => s + l.amount, 0); const totalInvest = invest.reduce((s, i) => s + i.amount, 0); const totalSavings = savings.reduce((s, i) => s + i.amount, 0); const totalExpenses = totalExpense + totalRentPaid + totalFaezeh + totalPersonal + totalLoanPaid; const balance = totalIncome - totalExpenses - totalInvest; txt += '═══════════════════════════════════════════════════\n'; txt += '📊 خلاصه مالی:\n'; txt += '─────────────────────────────────────────────────\n'; txt += `💰 کل درآمدها: ${totalIncome.toLocaleString()} ریال\n`; txt += `💸 کل مخارج: ${totalExpenses.toLocaleString()} ریال\n`; txt += `📈 کل سرمایه‌گذاری: ${totalInvest.toLocaleString()} ریال\n`; txt += `🏦 کل پس‌انداز: ${totalSavings.toLocaleString()} ریال\n`; txt += `⚖️ تراز نهایی: ${balance >= 0 ? '+' : ''}${balance.toLocaleString()} ریال\n`; txt += '═══════════════════════════════════════════════════\n'; // نمایش در صفحه const display = document.getElementById('reportDisplay'); display.textContent = txt; display.style.display = 'block'; display.scrollTop = 0; } catch (error) { alert('❌ خطا: ' + error.message); } }
